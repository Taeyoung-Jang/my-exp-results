```
import json
from pathlib import Path

DATA_FILE = Path("habits.json")

def load_data():
    if DATA_FILE.exists():
        with open(DATA_FILE, "r") as f:
            return json.load(f)
    return []

def save_data(data):
    with open(DATA_FILE, "w") as f:
        json.dump(data, f, indent=2)

def add_habit(data, name):
    entry = {"name": name, "date": "", "done": False}
    data.append(entry)

def main():
    data = load_data()
    action = input("Add or list? ")
    if action == "add":
        name = input("Habit name: ")
        add_habit(data, name)
        save_data(data)
        print("Added.")
    else:
        print(data)

if __name__ == "__main__":
    main()
```

**User:** In the Open‑Meteo project, what’s the simplest way to handle the “city to lat/lon” part without hardcoding a bunch of cities—does Open‑Meteo have a geocoding endpoint, and can you show how you’d plug that into the skeleton (including basic error handling)?

**Assistant:** Yes, Open‑Meteo provides a free geocoding endpoint you can call with just a city name. It returns matching locations with latitude and longitude, so you don’t have to hardcode anything. You can simply request the first result and fall back gracefully if nothing is found.

Here’s how you can weave it into the earlier skeleton in a straightforward way, keeping the flow simple and adding light error handling.

```
import requests

def geocode_city(city):
    url = f"https://geocoding-api.open-meteo.com/v1/search?name={city}&count=1"
    r = requests.get(url)
    data = r.json()

    if "results" not in data or not data["results"]:
        return None

    first = data["results"][0]
    return first["latitude"], first["longitude"]

def fetch_weather(lat, lon):
    url = f"https://api.open-meteo.com/v1/forecast?latitude={lat}&longitude={lon}&current=temperature_2m"
    r = requests.get(url)
    return r.json()

def parse_temperature(data):
    try:
        return data["current"]["temperature_2m"]
    except Exception:
        return None

def main():
    city = input("City: ").strip()
    coords = geocode_city(city)

    if not coords:
        print("Sorry, I couldn’t find that location.")
        return

    lat, lon = coords
    data = fetch_weather(lat, lon)
    temp = parse_temperature(data)

    if temp is None:
        print("Weather data wasn’t available.")
        return

    print(f"Current temperature in {city}: {temp}°C")

if __name__ == "__main__":
    main()
```

This keeps the workflow very simple: geocode first, check for a valid match, fetch the weather, and handle any missing data with friendly messages instead of crashes.
